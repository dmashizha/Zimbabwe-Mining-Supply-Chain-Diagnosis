// ── CONFIG — paste your Google Apps Script Web App URL here after setup ──────
const SHEET_URL = "https://script.google.com/macros/s/AKfycbxkxxBsR4Qzgs9i4ySfb2TEnOeQAqH5kNkl1q_fvkvD1vdPp0ZCmyNZqEFN-vzQN7D3/exec";

// ── Progress counter ──────────────────────────────────────────────────────────
function updateProgress(total) {
  const qs = document.querySelectorAll('.q-block');
  let answered = 0;
  qs.forEach(b => {
    const code = b.dataset.code;
    if (code && document.querySelector(`input[name="${code}"]:checked`)) answered++;
  });
  const el = document.getElementById('prog-label');
  if (el) el.textContent = answered + ' of ' + total + ' questions answered';
}

// ── Collect all form data into a flat object ──────────────────────────────────
function collectFormData() {
  const data = {};
  data['_submitted_at'] = new Date().toISOString();
  data['_form'] = document.title;

  const fd = new FormData(document.getElementById('form'));
  const multiFields = {};

  for (let [k, v] of fd.entries()) {
    if (multiFields[k]) {
      multiFields[k].push(v);
    } else if (data[k] !== undefined) {
      multiFields[k] = [data[k], v];
      delete data[k];
    } else {
      data[k] = v;
    }
  }
  // Join multi-select fields with semicolons
  for (let [k, vals] of Object.entries(multiFields)) {
    data[k] = vals.join('; ');
  }
  return data;
}

// ── Show submission status ────────────────────────────────────────────────────
function showStatus(msg, isError) {
  let el = document.getElementById('submit-status');
  if (!el) {
    el = document.createElement('div');
    el.id = 'submit-status';
    el.style.cssText = 'margin-top:10px;font-size:13px;padding:10px 14px;border-radius:4px;';
    document.querySelector('.submit-row').after(el);
  }
  el.textContent = msg;
  el.style.background = isError ? '#fce8e6' : '#e6f4ea';
  el.style.color = isError ? '#c5221f' : '#137333';
  el.style.display = 'block';
}

// ── Validate then submit ──────────────────────────────────────────────────────
function validateAndSubmit(e, total) {
  e.preventDefault();

  // Validate all rating questions
  const blocks = document.querySelectorAll('.q-block[data-code]');
  let valid = true, firstErr = null;
  blocks.forEach(b => {
    const code = b.dataset.code;
    if (!document.querySelector(`input[name="${code}"]:checked`)) {
      b.classList.add('error'); valid = false;
      if (!firstErr) firstErr = b;
    } else {
      b.classList.remove('error');
    }
  });
  if (!valid) {
    firstErr.scrollIntoView({ behavior: 'smooth', block: 'center' });
    return;
  }

  // Disable submit button and show sending state
  const btn = document.querySelector('.btn-submit');
  btn.disabled = true;
  btn.textContent = 'Submitting...';
  showStatus('Sending your responses to the database...', false);

  const data = collectFormData();

  // If no URL configured yet, just show thank you (development mode)
  if (!SHEET_URL || SHEET_URL === "https://script.google.com/macros/s/AKfycbxkxxBsR4Qzgs9i4ySfb2TEnOeQAqH5kNkl1q_fvkvD1vdPp0ZCmyNZqEFN-vzQN7D3/exec") {
    console.log('DEV MODE — responses not sent to sheet:', data);
    showThankYou();
    return;
  }

  // Send to Google Apps Script Web App
  fetch(SHEET_URL, {
    method: 'POST',
    mode: 'no-cors',          // required for Apps Script
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(data)
  })
  .then(() => {
    // no-cors means we can't read the response — assume success if no error thrown
    showThankYou();
  })
  .catch(err => {
    console.error('Submission error:', err);
    btn.disabled = false;
    btn.textContent = 'Submit';
    showStatus('Submission failed. Please check your connection and try again.', true);
  });
}

function showThankYou() {
  document.getElementById('form').style.display = 'none';
  document.querySelector('.progress-wrap') && (document.querySelector('.progress-wrap').style.display = 'none');
  document.getElementById('thankyou').style.display = 'block';
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

// ── Clear form ────────────────────────────────────────────────────────────────
function clearAll() {
  if (confirm('Clear all responses? This cannot be undone.')) {
    document.getElementById('form').reset();
    document.querySelectorAll('.q-block').forEach(b => b.classList.remove('error'));
    const s = document.getElementById('submit-status');
    if (s) s.style.display = 'none';
    updateProgress(53);
  }
}

// ── Wire up events ────────────────────────────────────────────────────────────
document.getElementById('form').addEventListener('change', () => updateProgress(53));
document.getElementById('form').addEventListener('submit', (e) => validateAndSubmit(e, 53));
